const v5BaseCollect=collect;
collect=function(){
  const chest=lootAt(S.p.x,S.p.y);
  if(chest&&chest.t==='chest'){
    S.loot=S.loot.filter(v=>v!==chest);
    S.mode='chest';
    S.msg+=' '+chest.name+' found — next roll opens it.';
    sfx('reward');
    return;
  }
  v5BaseCollect();
};

function chestReward(n){
  const names=['','OLD BLADE: ATK +1','TIN MAIL: ARM +1','MEAT PIE: +4 HP','COIN HOARD: +4 GOLD','ODD TALISMAN: LUCK +1','JACKPOT: +2 GOLD, FULL HEAL'];
  if(n===1)S.atk++;
  else if(n===2)S.arm++;
  else if(n===3)S.hp=Math.min(S.max,S.hp+4);
  else if(n===4)S.gold+=4;
  else if(n===5)S.luck++;
  else {S.gold+=2;S.hp=S.max}
  return names[n];
}

  }
  return v5Resolve(n,rollMode)
};

// ATTACK has a second use: if there is no adjacent enemy, smash one REVEALED adjacent trap.
// Trap XP is finite and deliberately gives XP but no treasure reward.
doAttack=function(bonus=0){
  const near=S.enemies.filter(e=>dist(S.p,e)===1).sort((a,b)=>a.hp-b.hp);
  if(!near.length){
    const trap=S.traps.find(t=>t.shown&&dist(S.p,t)===1);
    if(trap){
      S.traps=S.traps.filter(v=>v!==trap);
      const levels=metaGainXp(1);
      if(levels){S.mode='levelup';S.msg='Trap smashed. +1 XP. LEVEL '+S.level+' — next roll upgrades that face.'}
      else S.msg='Trap smashed. +1 XP.';
      sfx('attack');return;
    }
    S.msg='No Enemy or revealed Trap adjacent.';sfx('attack');return;
  }
  const e=near[0];let d=S.atk+bonus;if(S.luck>0&&Math.random()<.08*S.luck)d++;e.hp-=d;S.msg=e.name+' takes '+d+' damage.';sfx('attack');
  if(e.hp<=0){S.enemies=S.enemies.filter(v=>v!==e);const levels=metaGainXp(1);S.rewardQueued=true;if(levels){S.mode='levelup';S.msg=e.name+' defeated. LEVEL '+S.level+' — next roll upgrades that face.'}else{S.mode='reward';S.msg=e.name+' defeated — next roll is reward.'}sfx('reward')}
};
