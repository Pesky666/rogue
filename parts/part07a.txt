const SCORE_URL='https://roll-scores.ritual-x-daily.workers.dev/scores';

async function refreshTodaysBest(){
  try{
    const r=await fetch(SCORE_URL,{cache:'no-store'});
    if(!r.ok)throw new Error('scores '+r.status);
    const rows=await r.json();
    S.todayBest=Array.isArray(rows)?rows.slice(0,5).map(v=>Math.max(0,Math.floor(Number(v.gold)||0))):[];
    S.scoreStatus='ready';
  }catch(err){
    console.warn('Leaderboard fetch failed',err);
    S.scoreStatus='offline';
  }
  draw();
}

async function onGameOver(){
  if(!S||S.scoreSubmitted)return;
  S.scoreSubmitted=true;
  S.scoreStatus='loading';
  S.todayBest=[];
  draw();
  try{
    const r=await fetch(SCORE_URL,{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({gold:Math.max(0,Math.floor(S.gold||0))})
    });
    if(!r.ok)throw new Error('submit '+r.status);
  }catch(err){
    console.warn('Score submit failed',err);
  }
  await refreshTodaysBest();
}
