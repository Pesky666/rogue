function clearLOS(a,b){
  let x0=a.x,y0=a.y,x1=b.x,y1=b.y;
  let dx=Math.abs(x1-x0),sx=x0<x1?1:-1,dy=-Math.abs(y1-y0),sy=y0<y1?1:-1,err=dx+dy;
  while(!(x0===x1&&y0===y1)){
    const e2=2*err;if(e2>=dy){err+=dy;x0+=sx}if(e2<=dx){err+=dx;y0+=sy}
    if(x0===x1&&y0===y1)break;
    if(wall(x0,y0))return false;
  }
  return true;
}

  return (S.exit&&S.exit.x===x&&S.exit.y===y)||S.food.some(v=>v.x===x&&v.y===y)||S.coin.some(v=>v.x===x&&v.y===y)||S.items.some(v=>v.x===x&&v.y===y)||S.loot.some(v=>v.shown&&v.x===x&&v.y===y)||S.traps.some(v=>v.shown&&v.x===x&&v.y===y);
}
